# OS idle loop in __OSReschedule (0x8002E5EC): no runnable thread, interrupts enabled,
# spins on RunQueueBits until an interrupt handler readies a thread
idle    8002E724

# The game waits for its frames in flight (0x7FE0C670) to be drawn with interrupts enabled:
#   7FD46400: OSSleepThread(0x7FE0C6F4); 7FD46408: if (frames in flight > 0) goto 7FD46400
# Interrupts were taken at the back-edge, between the check and the sleep: a draw done's
# callback delivered there counted the last frame and woke a queue nobody slept on yet, and the
# game thread then slept for good (the freeze entering the attract demo or after a mission,
# issues #33 #36 #50; most runs under Android's arm64 translation, where the window is long). On
# hardware the window is a few instructions; without the poll the interrupt is taken inside
# OSSleepThread's reschedule, after the thread is on the queue. (The SDK's own loops of this
# shape, 0x8003C468 and 0x80043054, run with interrupts disabled.)
nopoll  7FD46400
